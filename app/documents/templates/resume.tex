<# The app's one-page resume. Every << value >> is escaped by app/documents/latex.py; this file is the only LaTeX. #>
\documentclass[letterpaper,10pt]{article}
\usepackage[left=0.4in,right=0.4in,top=0.3in,bottom=0.3in]{geometry}
\usepackage[hidelinks]{hyperref}
\usepackage{enumitem}
\setlist[itemize]{leftmargin=1.2em,topsep=1pt,itemsep=0pt,parsep=0pt}
\pagestyle{empty}
\setlength{\parindent}{0pt}
\newcommand{\resumesection}[1]{\vspace{5pt}{\large\bfseries #1}\par\vspace{-3pt}\rule{\linewidth}{0.4pt}\par\vspace{2pt}}
\begin{document}
\begin{center}
{\LARGE\bfseries << name >>}\par\vspace{2pt}
<% for field, value in header.items() if value %>
<% if not loop.first %>\enspace\textbullet\enspace <% endif %>
<% if field in ("email", "phone", "place") %><< value >><% else %>\href{<< value | href >>}{<< value | shown >>}<% endif %>
<% endfor %>
\end{center}
<% for section in sections %>
\resumesection{<< section.heading >>}
<% for block in section.blocks %>
<% if block.layout == "item" %>
\begin{itemize}
<% for tile in block.tiles %>
\item << tile.facts.name >>
<% endfor %>
\end{itemize}
<% else %>
<% for tile in block.tiles %>
<% set f = tile.facts %>
<% if block.layout == "skills" %>
\textbf{<< f.name >>:} << f.skills | join(", ") >>\par
<% elif block.layout == "paragraph" %>
<% for text in tile.bullets %>
<< text >>\par
<% endfor %>
<% else %>
\textbf{<< f.name >>}<% if f.rest %> --- << f.rest >><% endif %>\hfill << f.right >>\par
<% if f.second or f.second_right %>
\textit{<< f.second >>}\hfill \textit{<< f.second_right >>}\par
<% endif %>
<% if tile.bullets %>
\begin{itemize}
<% for text in tile.bullets %>
\item << text >>
<% endfor %>
\end{itemize}
<% endif %>
\vspace{2pt}
<% endif %>
<% endfor %>
<% endif %>
<% endfor %>
<% endfor %>
\end{document}
